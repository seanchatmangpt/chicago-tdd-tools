Test Macros

Synchronous Test

Requires: No feature flags. Core functionality.

File: tests/my_test.rs:

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

test!(test_sync, {
    // Arrange
    let input = 5;
    // Act
    let result = input * 2;
    // Assert
    assert_eq!(result, 10);
});
}

Async Test

Requires: Tokio runtime (included in dev-dependencies). No feature flags needed.

File: tests/my_async_test.rs:

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

async_test!(test_async_operation, {
    // Arrange
    let expected = 10;
    // Act
    let result = async {
        // Your async operation here
        tokio::time::sleep(tokio::time::Duration::from_millis(1)).await;
        5 * 2
    }.await;
    // Assert
    assert_eq!(result, expected);
});
}

Async Test with Custom Timeout

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

async_test_with_timeout!(test_async_integration, 30, {

Performance Test

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

performance_test!(test_hot_path, {
    // Arrange: Set up test data
    let input = vec![1, 2, 3];
    // Act: Execute hot path and measure ticks
    let (result, ticks) = measure_ticks(|| {
        input.iter().sum::<i32>()
    });
    // Assert: Verify performance constraint (≤8 ticks)
    assert_within_tick_budget!(ticks, "Hot path operation");
    assert_eq!(result, 6);
});
}

Parameterized Test

Requires: parameterized-testing feature flag.

Enable in Cargo.toml:

[dev-dependencies]
chicago-tdd-tools = { 
    path = "../chicago-tdd-tools",
    features = ["parameterized-testing"]
}

File: tests/my_param_test.rs:

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

param_test! {
    #[case(1, 2, 3)]
    #[case(2, 3, 5)]
    #[case(3, 4, 7)]
    fn test_addition(a: i32, b: i32, expected: i32) {
        assert_eq!(a + b, expected);
    }
}
}

Weaver Test

Requires: weaver feature flag (automatically enables otel).

Bootstrap once:

just weaver-bootstrap

Daily guardrail:

just weaver-smoke    # version check + telemetry span

Enable in Cargo.toml:

[dev-dependencies]
chicago_tdd_tools = { 
    path = "../chicago-tdd-tools",
    features = ["weaver"]  # Automatically enables otel
}

File: tests/my_weaver_test.rs:

#![allow(unused)]
fn main() {
use chicago_tdd_tools::{weaver_test, prelude::*};
use chicago_tdd_tools::observability::weaver::WeaverValidator;
use std::path::PathBuf;

weaver_test!(test_weaver_validation, {
    // Arrange: Create validator (infallible constructor; start() checks
    // binary/Docker/registry availability)
    let mut validator = WeaverValidator::new(PathBuf::from("registry/"));
    // Act: Start Weaver
    let start_result = validator.start();
    // Assert: Verify Weaver started or handle unavailable case
    if start_result.is_ok() {
        assert!(validator.is_running());
        assert!(validator.stop().is_ok());
    }
});
}

The validator lives at chicago_tdd_tools::observability::weaver::WeaverValidator (feature-gated: #[cfg(feature = "weaver")]). Key surface: new(PathBuf) (infallible), start(&mut self) -> WeaverValidationResult<()>, stop(&mut self), is_running(&self) -> bool, and WeaverValidator::check_weaver_available() for a binary-only availability probe.

Weaver Test with Custom Ports (requires weaver feature)

#![allow(unused)]
fn main() {
use chicago_tdd_tools::weaver_test;
use chicago_tdd_tools::observability::weaver::WeaverValidator;
use std::path::PathBuf;

weaver_test!(test_weaver_integration, {
    // Arrange: Create validator with custom OTLP gRPC and admin ports
    let mut validator =
        WeaverValidator::with_config(PathBuf::from("registry/"), 4318, 8081);
    // Act: Start Weaver
    let start_result = validator.start();
    // Assert: Verify Weaver started or handle unavailable case
    if start_result.is_ok() {
        assert_eq!(validator.otlp_endpoint(), "http://127.0.0.1:4318");
        assert!(validator.stop().is_ok());
    }
});
}

Alert Macros

Alert Levels

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

test!(test_alerts, {
    // Critical alert - must stop immediately
    alert_critical!("Docker daemon is not running", "Start Docker Desktop");
    
    // Warning alert - should stop
    alert_warning!("Container operation failed", "Check container state");
    
    // Info alert - informational
    alert_info!("Container started successfully");
    
    // Success alert - operation completed
    alert_success!("Test completed successfully");
    
    // Debug alert - detailed diagnostics
    alert_debug!("Container state: {:?}", "running");
    
    // Custom alert
    alert!("🚨", "Custom critical error", "STOP: Cannot proceed", "FIX: Resolve issue");
});
}

#[derive(TestBuilder)]

}

Async Fixtures (requires async feature, Rust 1.75+)

#![allow(unused)]
fn main() {
#[cfg(feature = "async")]
use chicago_tdd_tools::core::async_fixture::{AsyncFixtureManager, AsyncFixtureProvider};
use chicago_tdd_tools::core::fixture::FixtureError;

#[cfg(feature = "async")]
struct DatabaseFixture {
    connection: String,
}

#[cfg(feature = "async")]
struct DatabaseProvider;

#[cfg(feature = "async")]
AUDIT NOTE (W27 probe, compile-verified): `private::Sealed` is NOT publicly nameable — external crates cannot implement `AsyncFixtureProvider` (sealed by design). The snippet below is illustrative of the internal pattern only. Original: impl chicago_tdd_tools::core::async_fixture::private::Sealed for DatabaseProvider {}

#[cfg(feature = "async")]
impl AsyncFixtureProvider for DatabaseProvider {
    type Fixture<'a> = DatabaseFixture;
    type Error = FixtureError;

    async fn create_fixture(&self) -> Result<Self::Fixture<'_>, Self::Error> {
        Ok(DatabaseFixture { connection: "connected".to_string() })
    }
}

#[cfg(feature = "async")]
async_test!(test_async_fixture, {
    // Arrange: Create async fixture manager
    let provider = DatabaseProvider;
    let manager = AsyncFixtureManager::new(provider);
    // Act: Setup async fixture
    let fixture = manager.setup().await?;
    // Assert: Verify fixture created
    assert_eq!(fixture.connection, "connected");
    Ok::<(), FixtureError>(())
});
}

τ-Aware Thermal Testing (Track 2)

#![allow(unused)]
fn main() {
use chicago_tdd_tools::prelude::*;

// Hot path: τ ≤ 8 ticks (use relaxed config in tests)
let relaxed_config = HotPathConfig {
    max_ticks: 1000,
    enforce_no_alloc: false,
    enforce_no_syscall: false,
};
let hot_test = HotPathTest::new(relaxed_config);

let (value, ticks) = hot_test.run(|| critical_function()).unwrap();
assert!(ticks <= 1000); // In production: ticks <= 8
}
